🚀 QUICK START
────────────────────────────────────────────────────────────────────────────
1. Setup environment:
   cp .env.example .env
   # Edit .env and set POSTGRES_PASSWORD

2. Authenticate with Garmin:
   pip install garminconnect
   python3 -c "import garth; garth.login('email', 'pass'); garth.save('~/.garminconnect')"

3. Initialize:
   make setup

4. First sync:
   make sync-full

5. Automate:
   crontab -e
   # Add: 0 2 * * * /path/to/cron/sync-cron.sh

📝 USAGE EXAMPLES
────────────────────────────────────────────────────────────────────────────
# Incremental sync (daily)
docker-compose --profile sync run --rm garmin-sync

# Full sync (last 90 days)
docker-compose --profile sync run --rm garmin-sync --full

# Dry run
docker-compose --profile sync run --rm garmin-sync --dry-run

# Specific categories
docker-compose --profile sync run --rm garmin-sync --categories daily_health activities

# Custom date range
docker-compose --profile sync run --rm garmin-sync --start-date 2024-01-01 --end-date 2024-01-31

# Using Makefile
make sync              # Incremental
make sync-full         # Full sync
make sync-dry          # Dry run
make backup            # Backup DB
make status            # Check sync status
